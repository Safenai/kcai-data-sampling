1"""What one transformation output is, in memory. 

2 

3The row carries declarations, what replays the run, and a derived ``id`` — 

4never a judgement (a regime is read afterwards, from the input/output pair), 

5never a measurement (delta is recomputed downstream), never a path 

6(storage/IO adds it), never an *arbitrary* key: the id is "the fields it 

7would hash", materialized for consumers and lineage. 

8"""

10from dataclasses import dataclass 

11import hashlib 

12import json 

13from typing import Any

15import numpy as np

18def output_identity(fields: dict[str, Any]) -> str: 

19 """Derive the deterministic id of one output from its row fields. 

20 

21 Args: 

22 fields: The row fields (everything but the bitmap and the id itself). 

23 

24 Returns: 

25 A 12-hex sha1 over the fields, stable across runs and environments 

26 (``sort_keys``, ``default=str``): the id is re-derivable by any 

27 consumer, and two rows with the same id share the same recipe. 

28 """ 

29 identity = {k: v for k, v in fields.items() if k != "id"} 

30 return hashlib.sha1(json.dumps(identity, sort_keys=True, default=str).encode("utf-8")).hexdigest()[:12]

33@dataclass 

34class Output: 

35 """One output: the bitmap, and what could not be recovered any other way. 

36 

37 Lineage is a reference, ``parent_id``, into the selection the run was 

38 given. Measurements (delta, PSNR, ...) are recomputed downstream: ``params`` 

39 and ``seed`` replay the run. What the algorithm declares about itself 

40 (``family``, ``arity``, ``reversible``) is frozen here, because a lookup 

41 from ``algorithm`` gives what the code says *now*. 

42 

43 Attributes: 

44 x: The generated sample itself, a numpy array. 

45 id: The derived identity: a 12-hex sha1 of this output's row fields. 

46 parent_id: Lineage, a reference into the run's input selection. 

47 algorithm: The transformation's identity string. 

48 family: ``"procedural"``, ``"generative"`` or ``"adversarial"``. 

49 arity: ``"unary"`` or ``"n-ary"``. 

50 reversible: Whether ``x`` is determined by ``x'``, as declared. 

51 params: The fully resolved algorithm parameters, as given in the config. 

52 seed: The random seed, ``None`` for deterministic algorithms. 

53 tool_model: Name of the tool model used, or ``None``. 

54 target_model: Name of the target model used, or ``None``. 

55 """

72 @property 

73 def id(self) -> str: 

74 """The derived identity: a 12-hex sha1 of this output's row fields. 

75 

76 Not a stored field: it is recomputed from ``row()``, so it cannot 

77 drift from the row it identifies. 

78 """ 

79 return output_identity({k: v for k, v in vars(self).items() if k != "x"})

81 def row(self) -> dict[str, Any]: 

82 """Return every field but the bitmap, plus the derived ``id``. 

83 

84 Returns: 

85 A dictionary of the declarations and replay fields (everything 

86 except ``x``), with the derived ``id`` first and ``params`` kept 

87 as a ``dict`` (serialization to JSON is the writer's business). 

88 """ 

89 fields = {k: v for k, v in vars(self).items() if k != "x"} 

90 return {**fields, "id": output_identity(fields)}
