31class Transformation: 

32 """Shared identity and declarations for every transformation. 

33 

34 Lifecycle methods live in the arity subclasses, :class:`UnaryTransformation` 

35 and, later, :class:`NAryTransformation`. Construction resolves the 

36 parameters **through the algorithm's registered pydantic ``Config``** — the 

37 single declaration of parameters and defaults, so a bad config fails fast 

38 at load — and enforces that exactly the slot named by ``model_role`` is 

39 filled. 

40 

41 Attributes: 

42 algorithm: Set by every algorithm; the registry-resolved ``type``. 

43 family: The map's kind, ``None`` to derive it from the role of the 

44 model: ``"procedural"`` (no model), ``"generative"`` (tool) or 

45 ``"adversarial"`` (target). 

46 Config: The registered pydantic schema the algorithm's params are 

47 validated against; declared by every algorithm package. 

48 model_role: The model's role, ``None | "tool" | "target"``, which fixes 

49 the slot to fill. 

50 model_methods: The methods the model must expose (see ``api.roles``), 

51 checked at construction. 

52 stochastic: Does the algorithm draw randomness? Only then does a seed 

53 draw. 

54 arity: ``"unary"`` (``"n-ary"`` later), fixed by the base class. 

55 clips: May the output leave the selection's value range? A declaration: 

56 the base clips to the range only when this is set; an undeclared 

57 overflow is refused. 

58 reversible: Is ``x`` determined by ``x'``? A structural declaration, 

59 frozen on the row for the downstream judge. 

60 seed: The random seed; always accepted by the config, but only drawn 

61 when ``stochastic`` (relaxed seed rule). 

62 params: The fully resolved parameters, complete. 

63 tool_model: The tool model instance, or ``None``. 

64 target_model: The target model instance, or ``None``. 

65 """

108 algorithm parameter.

184 def fit_to_range(self, out: np.ndarray, value_range: tuple[float, float] | None) -> np.ndarray: 

185 """Clip if the algorithm declared it; then check. 

186 

187 Args: 

188 out: The algorithm's output batch. 

189 value_range: ``(low, high)`` domain of the selection, or ``None`` 

190 to leave the output unchecked. 

191 

192 Returns: 

193 The output, clipped when the algorithm declares ``clips``. 

194 

195 Raises: 

196 ValueError: If the output leaves the range and the algorithm did 

197 not declare ``clips``. 

198 """ 

199 if value_range is None: 

200 return out 

201 low, high = value_range 

202 if self.clips: 

203 clipped: np.ndarray = np.clip(out, low, high) 

204 if np.issubdtype(out.dtype, np.integer) and clipped.dtype != out.dtype: 

205 clipped = clipped.astype(out.dtype) 

206 return clipped 

207 lo, hi = float(out.min()), float(out.max()) 

208 if lo < low or hi > high: 

209 raise ValueError( 

210 f"{self.algorithm} left the selection's value range [{low}, {high}]: output in " 

211 f"[{lo:.4g}, {hi:.4g}]. An algorithm whose output can leave it declares `clips = True`." 

212 ) 

213 return out

215 def resolve(self, given: dict[str, Any]) -> dict[str, Any]:

217

243 def rngs(self, keys: list[str]) -> list[np.random.Generator] | None: 

244 """Yield one generator per output, seeded by seed, params and parent id(s). 

245 

246 Seeding from the output's parent ids makes a draw independent of both 

247 the batch and the order of the run. The resolved ``params`` are hashed 

248 in too, so the swept variants of one parent (one instance per value, 

249 same seed) each draw their own randomness. 

250 

251 Args: 

252 keys: One key per output (the parent id, or the joined parent ids 

253 for n-ary), used to derive each generator's seed. 

254 

255 Returns: 

256 One ``np.random.Generator`` per key, or ``None`` when the algorithm 

257 is deterministic. 

258 """ 

259 if not self.stochastic: 

260 return None 

261 assert self.seed is not None 

262 params_key = json.dumps(self.params, sort_keys=True, default=str) 

263 return [ 

264 np.random.default_rng( 

265 [ 

266 self.seed, 

267 int.from_bytes(hashlib.sha1(f"{k}::{params_key}".encode()).digest()[:8], "big"), 

268 ] 

269 ) 

270 for k in keys 

271 ]
