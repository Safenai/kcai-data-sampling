1"""Image codec helpers and payload naming. 

2 

3The RGBA-U8 pipeline: source files decode **once** to 

4raw RGBA bytes, flow through the engine as ``(B, H, W, 4)`` uint8 arrays, and 

5encode **once** to PNG at the outputs stage. Payload files are content-hashed 

6so identical generated images deduplicate. 

7 

8Hosted in ``-job``: every read happens in the dataloaders and every write in 

9the outputwriters of this package. 

10""" 

11 

12import hashlib 

13import io 

14from pathlib import Path 

15 

16import numpy as np 

17from PIL import Image 

18 

19SUPPORTED_EXTENSIONS = {".png", ".jpg", ".jpeg", ".bmp", ".webp"}

22def decode_image(path: str | Path) -> tuple[int, int, bytes]: 

23 """Decode an image file once into raw RGBA bytes. 

24 

25 Args: 

26 path: Path to an image file. 

27 

28 Returns: 

29 A ``(height, width, rgba_bytes)`` tuple where ``rgba_bytes`` is 

30 ``height * width * 4`` bytes of contiguous RGBA data (transparent 

31 pixels become alpha 0; opaque sources get alpha 255). 

32 

33 Raises: 

34 ValueError: If the file is not a decodable image. 

35 """ 

36 try: 

37 with Image.open(path) as img: 

38 rgba = img.convert("RGBA") 

39 return rgba.height, rgba.width, rgba.tobytes() 

40 except Exception as exc: 

41 raise ValueError(f"cannot decode image {path}: {exc}") from exc

44def encode_image(path: str | Path, array: np.ndarray) -> None: 

45 """Encode an HxW (3- or 4-channel) uint8 array as a PNG file. 

46 

47 Args: 

48 path: Destination path (parent dirs are created). 

49 array: ``(H, W, C)`` uint8 array. 

50 """ 

51 Path(path).parent.mkdir(parents=True, exist_ok=True) 

52 if array.dtype != np.uint8: 

53 raise ValueError("image arrays must be uint8") 

54 if array.ndim != 3 or array.shape[2] not in (3, 4): 

55 raise ValueError(f"expected (H, W, 3|4) array, got {array.shape}") 

56 Image.fromarray(np.ascontiguousarray(array)).save(path, format="PNG")

59def artifact_name(selection_name: str, output_id: str, x: np.ndarray) -> str: 

60 """Build the content-addressed payload file name for a row. 

61 

62 ``c6`` is a hash of the decoded payload bytes, so identical pixels plus 

63 the same output id name the same file (invariant artifact paths). The id 

64 already covers the recipe (parent, algorithm, settings, seed); ``c6`` only 

65 guards content collisions. 

66 

67 Args: 

68 selection_name: Name of the selection the row belongs to. 

69 output_id: The output's derived ``id``. 

70 x: The output ``(H, W, C)`` uint8 pixel array. 

71 

72 Returns: 

73 The ``.png`` artifact file stem to store under the payload directory. 

74 """ 

75 c6 = hashlib.sha1(np.ascontiguousarray(x).tobytes()).hexdigest()[:6] 

76 return f"{selection_name}__{output_id}__{c6}.png"

79def png_bytes(x: np.ndarray) -> bytes: 

80 """Encode an HxW uint8 array to PNG bytes without touching the disk. 

81 

82 Args: 

83 x: ``(H, W, 3|4)`` uint8 array. 

84 

85 Returns: 

86 The PNG file bytes. 

87 """ 

88 buffer = io.BytesIO() 

89 Image.fromarray(np.ascontiguousarray(x)).save(buffer, format="PNG") 

90 return buffer.getvalue()
