1"""The payload output writer: one hashed PNG payload file per row. 

2 

3Writes the pixel side of the metadata-only ledger: 

4``write_samples: false`` turns the run into a pure recipe trace (hashes only), 

5the payload files hold the pixels. 

6 

7Payload rows are encoded to bytes in memory and buffered; ``flush()`` writes 

8them. ``flush_batch_size`` bounds the buffer, so the writer never holds more 

9than that many encoded rows in memory. 

10"""

22class ImagesOutputWriter(OutputWriter): 

23 """Writes one encoded image file per output row. 

24 

25 The plugin for the ``images`` registered type 

26 (``kcai_data_sampling.outputwriter`` entry point). It owns the payload 

27 files only; the parquet writer owns the ledger, so this writer inherits 

28 the common writer protocol's no-op ``add_rows``. 

29 

30 Attributes: 

31 name: Unique writer name. 

32 """ 

33 

34 def __init__(self, name: str, config: dict[str, Any] | None = None): 

35 """Build the payload writer. 

36 

37 Args: 

38 name: Unique writer name. 

39 config: Writer configuration; ``samples_dir`` (payload directory 

40 pattern, ``{selection}`` substitutable), ``write_samples`` 

41 (emit files or trace-only) and ``flush_batch_size`` (payload 

42 rows buffered before a flush) are honored. 

43 """ 

44 config = config or {} 

45 super().__init__(name, config) 

46 self.samples_dir = config.get("samples_dir", "outputs/image/") 

47 self.write_samples = bool(config.get("write_samples", True)) 

48 self.flush_batch_size = config.get("flush_batch_size", 5) 

49 self._buffer: list[tuple[Path, bytes]] = [] 

50 

51 @override 

52 def add_payload(self, selection_name: str, output: Output) -> str | None: 

53 """Encode one output row to PNG bytes and buffer it. 

54 

55 The artifact name is content-addressed, so buffering and flush order 

56 cannot change the pixels on disk. No file is written until ``flush()``. 

57 

58 Args: 

59 selection_name: The selection the row belongs to; substituted 

60 into the ``samples_dir`` pattern. 

61 output: The output row; its ``x`` is the ``(H, W, 4)`` uint8 array. 

62 

63 Returns: 

64 The artifact file name (``{selection}__{id}__{c6}.png``). In 

65 trace-only mode the name is still returned (hash-only trace) but 

66 nothing is buffered. 

67 """ 

68 artifact = artifact_name( 

69 selection_name=selection_name, 

70 output_id=output.id, 

71 x=output.x, 

72 ) 

73 if self.write_samples: 

74 samples_dir = Path(self.samples_dir.format(selection=selection_name)) 

75 self._buffer.append((samples_dir / artifact, png_bytes(output.x))) 

76 if len(self._buffer) >= self.flush_batch_size: 

77 self.flush() 

78 return artifact 

79

82 """Ledger rows belong to the parquet writer; nothing to do here. 

83 

84 Args: 

85 selection_name: Ignored. 

86 rows: Ignored. 

87 """ 

88 del selection_name, rows

90 @override 

91 def flush(self) -> None: 

92 """Write all buffered payloads, then clear the buffer. 

93 

94 Payload files are written synchronously here — called by the job at 

95 selection end, and by the writer itself once the threshold is reached. 

96 """ 

97 for path, data in self._buffer: 

98 path.parent.mkdir(parents=True, exist_ok=True) 

99 path.write_bytes(data) 

100 self._buffer.clear()
