1"""Root job configuration. 

2 

3The input ``transformations`` entries are validated against each algorithm's 

4own registered pydantic schema inside :class:`SamplingInterfaceConfig`: core 

5cannot enumerate the discriminated union because algorithm packages arrive 

6after core, so the ``type`` string is looked up in the loaded transformation 

7registry and the entry is validated against the registered schema. Unknown or 

8mistyped types are refused at config load. 

9 

10A validated entry whose parameters carry ``SweepConfig`` values is *expanded* 

11at validation, before the registry consumers see it: each swept parameter 

12becomes as many concrete models as the values it covers (cartesian product 

13across several swept params), so the transformations list the job sees is 

14plain and long, and the id/params/artifact machinery is inherited 

15untouched. 

16 

17``ColumnsConfig`` and ``TransformationConfig`` live in 

18:mod:`kcai_data_sampling_core.models.transformation`; they are re-imported 

19here so ``from kcai_data_sampling_core.models.config import TransformationConfig`` 

20keeps working for algorithm packages and tests. 

21"""

36class JobConfig(BaseModel): 

37 """Root configuration for a sample-generation job. Each field maps to a pipeline stage. 

38 

39 Attributes: 

40 storage: Global storage config (local only). 

41 compute: Global compute / runtime settings. 

42 errors: Global error-handling policy. 

43 dataloaders: The data loaders and their selections. 

44 models: Named tool / target model references. 

45 operations: The single transform stage: the list of transformations 

46 (each applied independently to the same selection), outputs, and 

47 the batch/error/storage overrides. 

48 """ 

49 

50 model_config = ConfigDict(extra="forbid") 

51 

52 storage: StorageConfig | None = None 

53 compute: ComputeConfig | None = None 

54 errors: ErrorsConfig | None = None 

55 dataloaders: DataLoadersConfig = Field(description="Data loaders.") 

56 models: ModelsConfig | None = None 

57 operations: SamplingInterfaceConfig = Field(description="The single transformations interface.")

59 @model_validator(mode="after") 

60 def _expand_sweeps(self) -> Self: 

61 """Expand any ``SweepConfig``-valued parameters at config validation. 

62 

63 The sweep expansion happens here, once the ``operations`` and 

64 ``compute`` fields are validated: each swept transformation is replaced 

65 by as many concrete models as its values cover (cartesian product when 

66 several parameters sweep), seeded from ``compute.seed`` so ``mode: 

67 random`` draws are reproducible. The sweep has no seed of its own. 

68 

69 Returns: 

70 The complete job config with sweeps expanded. 

71 

72 Raises: 

73 ValueError: If a sweep is void or ``mode: random`` has no seed. 

74 """ 

75 seed = self.compute.seed if self.compute is not None else ComputeConfig().seed 

76 expanded: list[TransformationConfig] = [] 

77 for entry in self.operations.transformations: 

78 expanded.extend(expand_sweeps(entry, seed)) 

79 self.operations.transformations = expanded 

80 return self

82 @model_validator(mode="after") 

83 def _check_columns_input(self) -> Self: 

84 """Refuse a reserved ``columns.input`` that does not name a sample column. 

85 

86 ``columns.input`` is optional and, when present, must be a 

87 list of exactly one value — a loader's ``sample_path.column`` (the 

88 single processable sample column each row carries at load). The value 

89 is not consumed; the transformation reads the batch sample 

90 stack as always. The check runs after sweeps expanded, so every 

91 concrete variant is covered. 

92 

93 Returns: 

94 The validated job config. 

95 

96 Raises: 

97 ValueError: If ``columns.input`` names more than one column, or 

98 names a column no loader declares as its ``sample_path.column``. 

99 """ 

100 used = [t for t in self.operations.transformations if t.columns is not None and t.columns.input is not None] 

101 if not used: 

102 return self 

103 sample_columns = {sp.column for loader in self.dataloaders.loaders for sp in (loader.sample_path or [])} 

104 for entry in used: 

105 columns = entry.columns 

106 assert columns is not None 

107 input_columns = columns.input 

108 assert input_columns is not None 

109 if len(input_columns) != 1: 

110 raise ValueError(f"{entry.name}: columns.input must name exactly one column, got {input_columns!r}") 

111 if input_columns[0] not in sample_columns: 

112 raise ValueError( 

113 f"{entry.name}: columns.input {input_columns[0]!r} is not any loader's " 

114 f"sample_path.column (loaders: {sorted(sample_columns) or 'none declared'})" 

115 ) 

116 return self
